name: Refresh official fleet data

on:
  workflow_dispatch:
  schedule:
    - cron: '17 */6 * * *'

permissions:
  contents: write

concurrency:
  group: fleet-refresh
  cancel-in-progress: false

jobs:
  refresh:
    runs-on: ubuntu-latest
    timeout-minutes: 20
    env:
      TZ: UTC
      SEC_USER_AGENT: DrillshipStatus/1.0 contiloop@users.noreply.github.com
    steps:
      - name: Checkout exact main revision
        uses: actions/checkout@11d5960a326750d5838078e36cf38b85af677262
        with:
          fetch-depth: 0

      - name: Set up Python
        uses: actions/setup-python@a26af69be951a213d495a4c3e4e4022e16d87065
        with:
          python-version: '3.12'
          cache: pip
          cache-dependency-path: automation/requirements.txt

      - name: Set up Node
        uses: actions/setup-node@49933ea5288caeca8642d1e84afbd3f7d6820020
        with:
          node-version: '22'
          cache: npm

      - name: Install dependencies
        run: |
          python -m pip install --requirement automation/requirements.txt
          npm ci

      - name: Run parser and unit tests
        run: python -m pytest automation/tests -q

      - name: Discover, parse, validate, and publish generated JSON
        run: python -m automation.fleet_sync --write

      - name: Build dashboard against generated data
        run: npm run build

      - name: Verify generated-only diff
        id: generated
        shell: bash
        run: |
          changed="$(git status --porcelain=v1 --untracked-files=all -- public/data data/provenance)"
          if [ -z "$changed" ]; then
            echo "changed=false" >> "$GITHUB_OUTPUT"
            exit 0
          fi

          unexpected="$(git status --porcelain=v1 --untracked-files=all | cut -c4- | grep -Ev '^(public/data/|data/provenance/)' || true)"
          if [ -n "$unexpected" ]; then
            echo "Unexpected files changed:"
            echo "$unexpected"
            exit 1
          fi
          echo "changed=true" >> "$GITHUB_OUTPUT"

      - name: Commit validated refresh
        if: steps.generated.outputs.changed == 'true'
        shell: bash
        run: |
          remote_sha="$(git ls-remote origin refs/heads/main | cut -f1)"
          if [ "$remote_sha" != "$GITHUB_SHA" ]; then
            echo "main changed during refresh; aborting instead of publishing a stale merge"
            exit 1
          fi
          git config user.name "contiloop"
          git config user.email "126137639+contiloop@users.noreply.github.com"
          git add public/data data/provenance
          git commit -m "Refresh verified fleet data"
          git push origin HEAD:main
